--        Whether Elsa knows about Bolton's account discrepancies

--        Whether the room door is locked
--  4  -> Claire 是否持有副管家的亲笔信
--        Whether Claire holds the deputy manager's letter
--  5  -> Victor 是否在停电时看见某可疑身影
--        Whether Victor saw a suspicious figure during the blackout
--  6  -> Hamilton 是否确定副管家的死亡时间
--        Whether Hamilton has determined the deputy manager's time of death
--  7  -> Stone 是否拥有有效的旧主遗嘱副本
--        Whether Stone possesses a valid copy of the old will
--  8  -> Elsa 的财务危机是否暴露
--        Whether Elsa's financial crisis is exposed
--  9  -> Victor 是否协助 Stone
--        Whether Victor assists Stone
--  10 -> 是否发现毒药痕迹

--        Whether the deputy manager publicly mentioned Stone's crimes

--        Whether Claire knows about Stone's financial fraud
--  13 -> Stone 是否伪造密室证据
--        Whether Stone fabricated locked-room evidence

--        Whether Bolton holds the key to the secret passage
--  15 -> 是否发现副管家身上的旧收据
--        Whether an old receipt was found on the deputy manager
--  16 -> Stone 是否试图嫁祸 Claire
--        Whether Stone tried to frame Claire
--  17 -> 房门是否被外部反锁
--        Whether the door was locked from the outside

-- 定义规则 (LAW)
-- Define Rules
--========================================================
LAW
  (
    (1 -> ~2) &                  -- Elsa 知道账册异常，则 Bolton 无法得知遗嘱
                                -- If Elsa knows about account discrepancies, Bolton cannot know about the will
    (4 -> (13 & 16)) &           -- Claire 持有亲笔信则 Stone 试图伪造密室嫁祸
                                -- If Claire holds the letter, Stone attempts to fabricate a locked room and frame
    (7 -> (6 -> 3)) &            -- Stone 拥有有效遗嘱，则死亡时间与门锁相关
                                -- If Stone has a valid will, the time of death relates to the locked door
    (8 -> (1 & ~7)) &            -- Elsa 财务危机暴露时，她发现账册异常但未获得遗嘱
                                -- If Elsa's financial crisis is exposed, she finds the account discrepancies but not the will
    (9 -> ~3) &                  -- Victor 协助 Stone 时，房门未被锁
                                -- If Victor assists Stone, the door is not locked
    (10 -> (5 & ~6)) &           -- 毒药痕迹存在，则 Victor 看见但 Hamilton 无法确定死亡时间
                                -- If poison traces exist, Victor sees but Hamilton cannot determine the time of death
    (11 -> ~8) &                 -- 副管家提到 Stone 的犯罪行为，则 Elsa 的危机未暴露
                                -- If the deputy manager mentions Stone's crimes, Elsa's crisis is not exposed
    (12 -> 11) &                 -- Claire 知道 Stone 的欺诈，则副管家必提到
                                -- If Claire knows about Stone's fraud, the deputy manager must have mentioned it
    (13 -> 3) &                  -- 密室伪造则房门上锁
                                -- If the locked room is fabricated, the door is locked
    (14 -> 9) &                  -- Bolton 持有暗道钥匙，则 Victor 必协助 Stone
                                -- If Bolton holds the secret passage key, Victor must assist Stone
    (15 -> ~4)                   -- 如果发现旧收据，则 Claire 未持有亲笔信
                                -- If an old receipt is found, Claire does not hold the letter
  )

OBS
  elsa:     1,8,13                -- Elsa 能观察到账册异常、危机和伪造证据
                                 -- Elsa observes account discrepancies, crisis, and fabricated evidence
  bolton:   2,9,14                -- Bolton 能观察到遗嘱状态、Victor 和钥匙
                                 -- Bolton observes the will, Victor, and the key
  claire:   4,12,16               -- Claire 能观察到亲笔信、欺诈和嫁祸行为
                                 -- Claire observes the letter, fraud, and framing attempts
  victor:   5,9,10                -- Victor 能观察到可疑身影、协助和毒药痕迹
                                 -- Victor observes suspicious figures, assistance, and poison traces
  hamilton: 6,11                  -- Hamilton 能观察到死亡时间和犯罪提及
                                 -- Hamilton observes the time of death and crimes mentioned
  stone:    3,7,15                -- Stone 能观察到房门状态、有效遗嘱和旧收据
                                 -- Stone observes the door status, valid will, and old receipt

-- 定义查询 (Queries)
-- Define Queries
--========================================================

-- 1) Elsa 是否知道她的财务危机和 Stone 的欺诈？
-- Does Elsa know about her financial crisis and Stone's fraud?
TRUE?
  {}
  elsa knows that (8 & ~7)

-- 2) Claire 是否知道自己被嫁祸？
-- Does Claire know she is being framed?
TRUE?
  {}
  claire knows whether 16

-- 3) 哪些状态满足：(毒药存在 & Victor 在现场 & 房门未锁)
-- What states satisfy: (Poison exists, Victor is on-site, and the door is unlocked)?
WHERE?
  (10 & 5 & ~3)

-- 4) Stone 是否必然知道密室证据无效？
-- Does Stone inevitably know that the locked-room evidence is invalid?
VALID?
  [ ! (13 & 16) ]
  (stone knows that ~3)

-- 5) Hamilton 的时间线推断是否与毒药线索一致？
-- Does Hamilton's timeline match the poison trace evidence?
VALID?
  [ ! (10 & 6) ]
  (hamilton knows whether 3)
